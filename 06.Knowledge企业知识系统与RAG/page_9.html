🚨 多数据源冲突陷阱
Spring AI 默认会将自动装配的 PgVectorStore 绑定到项目的主数据源。如果业务库用的是 MySQL，启动时会因向 MySQL 写入 PG 向量列而直接崩溃。项目中必须显式定义一个独立的 PostgreSQL DataSource，并手动构建对应的 PgVectorStore Bean。
💻 Demo 代码
// VectorStoreConfig.java — 独立数据源
@Bean
public DataSource pgvectorDataSource() {
  HikariDataSource ds = new HikariDataSource();
  ds.setJdbcUrl(pgUrl);  // PostgreSQL
  ds.setUsername(pgUser);
  ds.setPassword(pgPass);
  ds.setMaximumPoolSize(5);
  return ds;
}

// PgVectorStore Bean
@Bean
public PgVectorStore pgVectorStore(
    JdbcTemplate pg, EmbeddingModel emb) {
  return PgVectorStore.builder(pg, emb).build();
}

// schema-knowledge.sql
CREATE TABLE vector_store (
  id        VARCHAR(255) PRIMARY KEY,
  content   TEXT NOT NULL,
  metadata  JSONB,
  embedding vector(1024) NOT NULL
);
CREATE INDEX idx_vs_embedding
  ON vector_store
  USING hnsw (embedding vector_cosine_ops);
