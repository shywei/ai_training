💻 v1.3 索引与检索核心代码（KnowledgeService + RetrievalService）
// 1. KnowledgeController: POST /api/knowledge/index → KnowledgeService.indexProducts()
public int indexProducts() {
  List<Document> docs = loadProductDocuments(); // 读取 products.json (26条)
  vectorStore.add(docs);                        // PgVectorStore 批量 Embedding 入库
  return docs.size();
}

// 2. RetrievalService.retrieve(query) — Spring AI SearchRequest 检索
public String retrieve(String query) {
  if (!isProductRelated(query)) return null;
  SearchRequest req = SearchRequest.query(query)
      .withTopK(5).withSimilarityThreshold(0.5); // 可配置 Top-K 与阈值(0.3~0.5)
  List<Document> docs = vectorStore.similaritySearch(req);
  return formatResults(docs);
}
