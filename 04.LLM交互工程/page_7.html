💻 Demo 代码：ModelRouter.java
@Component
public class ModelRouter {

  @Value("${agent.model.fast}")
  private String fastModel;

  @Value("${agent.model.reasoning}")
  private String reasoningModel;

  public String selectModel(String msg) {
    if (needsReasoning(msg)) {
      return reasoningModel;  // 复杂→推理模型
    }
    return fastModel;         // 简单→快模型
  }

  private boolean needsReasoning(String msg) {
    if (msg.length() > 500) return true;
    String[] kw = {"架构","重构","分析"};
    for (String k : kw)
      if (msg.contains(k)) return true;
    return false;
  }
}
