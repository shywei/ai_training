async sendMessage() {
  const response = await fetch(
    `/api/agent/chat?message=${text}`
  );

  // 读取 SSE 流
  const reader = response.body
    .getReader();
  const decoder = new TextDecoder();

  while (true) {
    const {done, value} =
      await reader.read();
    if (done) break;

    buffer += decoder.decode(value,
      { stream: true });

    // 解析 SSE: data: xxx
    for (const line of lines) {
      if (line.startsWith('data:')) {
        const content =
          line.substring(5);
        if (content === '[DONE]')
          continue;
        fullText += content;
        // 更新 UI
        msg.content = fullText;
      }
    }
  }
}
