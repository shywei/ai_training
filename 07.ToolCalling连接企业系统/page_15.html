McpProductClient.java — Tool 发现 + 包装
@PostConstruct
public void init() {
  // 1. 连接 MCP Server
  mcpClient = McpClient.sync(transport)
    .requestTimeout(Duration.ofSeconds(10))
    .build();
  mcpClient.initialize();

  // 2. 发现 Tool
  ListToolsResult tools = mcpClient.listTools();

  // 3. 包装为 FunctionCallback
  for (Tool tool : tools.tools()) {
    FunctionCallback cb = FunctionCallbackWrapper
      .builder(function)
      .withName("mcp_" + tool.name())
      .withDescription(tool.description())
      .build();
    callbacks.add(cb);
  }
}
