public Route route(String userMessage) {
  String prompt = """
    判断用户消息的意图，只输出一个词：
    - KNOWLEDGE：查询商品信息、推荐
    - TOOL：查询价格、库存、下单
    - CHAT：普通聊天

    用户消息：%s
    只输出 KNOWLEDGE、TOOL 或 CHAT。
    """.formatted(userMessage);

  String result = ChatClient.create(chatModel)
    .prompt().user(prompt)
    .options(OpenAiChatOptions.builder()
      .withModel(fastModel)
      .withTemperature(0.0)
      .build())
    .call().content();

  if (result.contains("TOOL")) return TOOL;
  if (result.contains("KNOWLEDGE")) return KNOWLEDGE;
  return CHAT;
}
Demo 已覆盖 · temperature=0 保证确定性输出
🚨 确定性路由与优雅降级
分类意图判定时，必须强制指定 temperature=0，否则随机性的输出容易产生意图颠倒。同时，在后端获取 mcpClient 绑定的 callback 时需做空指针防御与降级拦截，防止微服务启动顺序导致的进程挂起崩溃。

// ShoppingAgentController.java
Route route = llmRouter.route(message);

// context = 画像 + 记忆 + 历史（始终构建）
String context = memoryManager
  .buildContext(sessionId, userId, message);

// 仅 KNOWLEDGE 路由注入 RAG
if (route == Route.KNOWLEDGE) {
  String rag = mcpKnowledgeClient
    .search(message);
  context += "## 参考知识\n" + rag;
}

// 仅 TOOL 路由注册 product tools
if (route == Route.TOOL) {
  clientBuilder.functions(
    mcpProductClient.getFunctionCallbacks());
}

// save_memory 跨路由注册
clientBuilder.functions(
  mcpKnowledgeClient.getFunctionCallbacks());
