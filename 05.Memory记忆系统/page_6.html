Session Memory — Demo 实现

Demo 选择：Redis 方案
原因：支持多实例部署，自带过期机制，生产环境主流。
# Redis Key 设计
session:{sessionId}:messages  -> List

# 每条消息
"user|我叫张三"
"assistant|你好张三！"

# 操作
RPUSH  -> 追加 | LRANGE -> 读取
LPOP   -> 裁剪 | EXPIRE -> 24h过期
SessionMemory.java
@Component
public class SessionMemory {
  private final StringRedisTemplate redis;
  private static final int MAX = 20;

  public void addMessage(String sid,
      String role, String content) {
    String key = "session:" + sid + ":messages";
    redis.opsForList().rightPush(key,
        role + "|" + content);
    redis.expire(key, 24, TimeUnit.HOURS);
    if (redis.opsForList().size(key) > MAX)
      redis.opsForList().leftPop(key);
  }
}
